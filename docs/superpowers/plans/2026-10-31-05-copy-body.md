# REQ-05 一键复制正文 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 在详情页每行与编辑页一键把「标题 + 正文」（或仅标题 / 仅正文）转成可直接粘贴到作家后台的纯文本并写入剪贴板。

**Architecture:** 转换规则已在 REQ-02 的 `publish.ts`（`toPublishBody`/`buildPublishForm`）中**只有一份实现**；本方案只增加 `formatForCopy`（组装标题+正文）、服务端读盘取文本的 `getCopyPayload`、提示文案函数 `copyNotice`、浏览器剪贴板工具 `copyAsync`（含降级），以及三处 UI：详情页「复制 ▾」、编辑页「复制标题+正文」、设置页只读说明。复制是只读操作，不改库、不写文件、不改状态。

**Tech Stack:** TypeScript、Clipboard API（`ClipboardItem` 异步 Blob，降级 `execCommand`）、Vitest、Python 3（仅用于一次性生成对拍夹具）。前置：基础设施、REQ-01–03 已完成。

## Global Constraints

- 输出结构：`{第N章 标题}` + 空行 + `{正文}`；「仅标题」= `第N章 标题`；「仅正文」= 去标记后的正文；**正文不重复标题行**（首个 H1 已被移除）。
- 标题 = `第${seq}章 ${标题}`，`seq` 取文件名序号，标题取 H1（无 H1 时取 `chapter.title`）。
- 复制是只读：不标记已发、不写库、不写文件（F5.8）。
- 详情页复制必须**读取磁盘最新文件**，而非库中缓存（F5.4）；编辑页复制 **textarea 当前内容**，有未保存修改时提示「复制的是未保存内容」（F5.3）。
- `file_missing` 的章节：复制按钮置灰，`title="文件丢失"`（F5.7）。
- 转换规则以番茄发布技能 `pipeline.py` 的 `draft_chapter` 为基准。**已验证**：用 `draft_chapter` 与本项目 `toPublishBody` 对 `陈波宁传/改稿` 的 25 章真实稿件逐章比对，正文 25/25 一致。
- 与基准有意的差异（均为 `draft_chapter` 未覆盖的 Markdown，真实稿件中未出现）：保留行首全角缩进（Q7）；`#` 小标题、`*斜体*`、`~~删除线~~`、链接、图片、HTML 注释、转义会被处理；孤立 `*` 保留（`draft_chapter` 会把所有 `**` 直接删除）。
- 文案（逐字）：成功 `已复制《第N章 标题》，共 {N} 字`；仅标题 `已复制标题：第N章 标题`；正文为空 `已复制标题，正文为空`；失败 `无法写入剪贴板，请手动全选复制`；编辑页有未保存修改 `已复制，复制的是未保存内容`；字数口径同 REQ-02（发布形态正文去空白后的码点数）。

```ts
export type CopyMode = 'both' | 'title' | 'body';

export function formatForCopy(markdown: string, seq: number, fallbackTitle: string, mode: CopyMode = 'both'): string {
  const { title, body } = buildPublishForm(markdown, seq, fallbackTitle);
  if (mode === 'title') return title;
  if (mode === 'body') return body;
  return body ? `${title}\n\n${body}` : title;
}
```

```ts
// src/lib/copy-notice.ts  （纯函数，客户端可引用）
import type { CopyMode } from './publish';

export function copyNotice(
  p: { title: string; wordCount: number; bodyEmpty: boolean; mode: CopyMode },
  unsaved = false,
): string {
  if (unsaved) return '已复制，复制的是未保存内容';
  if (p.mode === 'title') return `已复制标题：${p.title}`;
  if (p.bodyEmpty) return '已复制标题，正文为空';
  return p.mode === 'body' ? `已复制《${p.title}》正文，共 ${p.wordCount} 字` : `已复制《${p.title}》，共 ${p.wordCount} 字`;
}
```

```ts
// src/lib/copy.ts  （服务端）
import { eq } from 'drizzle-orm';
import type { Db } from '@/db';
import { chapter } from '@/db/schema';
import { readChapterFile } from './file-io';
import { buildPublishForm, countWords, formatForCopy, type CopyMode } from './publish';
import { UserError } from './result';

export interface CopyPayload {
  text: string;
  title: string;
  bodyEmpty: boolean;
  wordCount: number;
  mode: CopyMode;
}

export function getCopyPayload(db: Db, chapterId: number, mode: CopyMode): CopyPayload {
  const ch = db.select().from(chapter).where(eq(chapter.id, chapterId)).get();
  if (!ch) throw new UserError('章节不存在');
  if (ch.fileMissing) throw new UserError('文件丢失，无法复制');
  let read;
  try {
    read = readChapterFile(ch.filePath);
  } catch (e) {
    throw new UserError(`无法读取文件：${(e as Error).message}`);
  }
  if (!read.ok) throw new UserError('编码不支持：文件不是 UTF-8，无法复制');
  const { title, body } = buildPublishForm(read.text, ch.seq, ch.title);
  return {
    text: formatForCopy(read.text, ch.seq, ch.title, mode),
    title,
    bodyEmpty: body === '',
    wordCount: countWords(body),
    mode,
  };
}
```

```ts
// src/lib/clipboard.ts  （仅浏览器端使用）
function legacyCopy(text: string): boolean {
  const ta = document.createElement('textarea');
  ta.value = text;
  ta.style.position = 'fixed';
  ta.style.opacity = '0';
  document.body.appendChild(ta);
  ta.select();
  try {
    return document.execCommand('copy');
  } catch {
    return false;
  } finally {
    ta.remove();
  }
}

export async function copyAsync(getText: () => Promise<string>): Promise<{ ok: boolean; text: string }> {
  const textPromise = getText();
  textPromise.catch(() => {}); // 避免第一条路径失败时出现未处理的 rejection；真正的异常在下面 await 时抛出
  if (typeof ClipboardItem !== 'undefined' && navigator.clipboard?.write) {
    try {
      await navigator.clipboard.write([
        new ClipboardItem({ 'text/plain': textPromise.then((t) => new Blob([t], { type: 'text/plain' })) }),
      ]);
      return { ok: true, text: await textPromise };
    } catch {
      /* 继续降级 */
    }
  }
  const text = await textPromise;
  try {
    await navigator.clipboard.writeText(text);
    return { ok: true, text };
  } catch {
    return { ok: legacyCopy(text), text };
  }
}
```

```tsx
// src/components/copy-menu.tsx
'use client';
import { useState } from 'react';
import { copyChapterAction } from '@/app/actions/copy';
import { copyAsync } from '@/lib/clipboard';
import { copyNotice } from '@/lib/copy-notice';
import type { CopyPayload } from '@/lib/copy';
import type { CopyMode } from '@/lib/publish';
import { useNotice } from './notice';

export function CopyMenu({ chapterId, fileMissing }: { chapterId: number; fileMissing: boolean }) {
  const [open, setOpen] = useState(false);
  const [done, setDone] = useState(false);
  const [busy, setBusy] = useState(false);
  const [fallback, setFallback] = useState<string | null>(null);
  const { notify } = useNotice();

  const run = async (mode: CopyMode) => {
    setBusy(true);
    setFallback(null);
    const holder: { payload?: CopyPayload } = {};
    try {
      const r = await copyAsync(async () => {
        const res = await copyChapterAction(chapterId, mode);
        if (!res.ok) throw new Error(res.error);
        holder.payload = res.data;
        return res.data.text;
      });
      if (!r.ok) {
        setFallback(r.text);
        return notify({ kind: 'error', text: '无法写入剪贴板，请手动全选复制' });
      }
      setOpen(false);
      setDone(true);
      setTimeout(() => setDone(false), 1500);
      notify({ kind: 'success', text: copyNotice(holder.payload!) });
    } catch (e) {
      notify({ kind: 'error', text: (e as Error).message });
    } finally {
      setBusy(false);
    }
  };

  if (fileMissing) {
    return <button className="rounded border px-3 text-stone-400" disabled title="文件丢失">复制 ▾</button>;
  }
  return (
    <div className="relative inline-block">
      <div className="inline-flex overflow-hidden rounded border">
        <button className="px-3" disabled={busy} onClick={() => run('both')}>{done ? '已复制 ✓' : '复制'}</button>
        <button className="border-l px-2" aria-label="更多复制方式" aria-expanded={open} onClick={() => setOpen(!open)}>▾</button>
      </div>
      {open && (
        <div className="mt-1 flex gap-2">
          <button className="rounded border px-3 text-sm" disabled={busy} onClick={() => run('title')}>仅标题</button>
          <button className="rounded border px-3 text-sm" disabled={busy} onClick={() => run('body')}>仅正文</button>
        </div>
      )}
      {fallback !== null && (
        <textarea readOnly className="mt-1 h-24 w-64 rounded border p-2 text-xs" value={fallback} onFocus={(e) => e.currentTarget.select()} />
      )}
    </div>
  );
}
```
