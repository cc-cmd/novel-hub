## Global Constraints

- 文件是真相，数据库是索引：章节正文永不入库；平台从不创建/修改/删除小说目录，章节文件只在「编辑页保存」时写入。
- 状态 `status` 仅三种入库值：`unpublished` / `pending_review` / `published`；「需更新」「文件丢失」是**计算态**，不入库。
- 需更新判定（唯一定义）：`status ∈ {pending_review, published}` 且 `published_hash ≠ chapter.content_hash`；`file_missing` 优先级最高；`content_hash` 为空不判需更新；`published_hash` 为空而状态为审核中/已发布按需更新处理。
- 字数 = 发布形态正文去掉全部空白后的字符数（按 Unicode 码点计），不含标题。
- 发布形态 hash = `sha256_hex(平台标题 + "\n" + 发布形态正文)`，平台标题 = `第${seq}章 ${标题}`，标题优先取 H1，缺失取文件名标题。
- 章节文件名格式 `^ch(\d+)-(.+)\.md$`（忽略大小写，半角 `-`），章节身份 = `(novel_id, seq)`，不以路径为身份。
- 发布记录（`publication`）受保护：同步、解绑、文件丢失、归档、移除根目录都不得删除它。
- 内联展开优先于弹窗；操作结果用页内提示条；取消发布状态的提示条带「撤销」。
- 字体：书名/编辑区 Noto Serif SC，界面 Noto Sans SC，章号/数字/路径 JetBrains Mono。可点击区域 ≥ 44px。
- 徽标（颜色+图标+文字）：○ 未发、◷ 审核中、✓ 已发、↻ 需更新、! 文件丢失。
- 页面层 MVP 仅手动验证；`fs-scan`、`hash`、`publish` 以及各 service 必须有 Vitest 单测。
- 偏差声明（相对设计文档）：不使用 shadcn CLI，统一用 Tailwind 原子类手写组件，以保证任务可确定地执行；若日后需要 shadcn 可逐个替换。

## 共享类型契约（所有方案共同遵守，名称不得改动）

```ts
// src/lib/result.ts
export type ActionResult<T = void> = { ok: true; data: T } | { ok: false; error: string };
export const ok = <T>(data: T): ActionResult<T> => ({ ok: true, data });
export const fail = (error: string): ActionResult<never> => ({ ok: false, error });

// src/lib/types.ts
export type PublishStatus = 'unpublished' | 'pending_review' | 'published';
export type Badge = 'unpublished' | 'pending_review' | 'published' | 'stale' | 'missing';
export type ChapterWarning = 'h1_mismatch' | 'no_h1' | 'encoding' | null;
```

- Service 函数签名模式：`fn(db: Db, ...args)`，`Db` 来自 `src/db/index.ts`。
- 异常约定：service 层对**可预期的业务失败**抛 `new UserError(message)`（`src/lib/result.ts` 导出）；Server Action 用 `wrap()` 把它转成 `fail(message)`，其他异常继续抛出。

```ts
// src/lib/result.ts
export type ActionResult<T = void> = { ok: true; data: T } | { ok: false; error: string };
export const ok = <T>(data: T): ActionResult<T> => ({ ok: true, data });
export const fail = (error: string): ActionResult<never> => ({ ok: false, error });

export class UserError extends Error {}

/** Server Action 包装：把 UserError 转成 fail，其余异常原样抛出。 */
export async function wrap<T>(fn: () => T | Promise<T>): Promise<ActionResult<T>> {
  try {
    return ok(await fn());
  } catch (e) {
    if (e instanceof UserError) return fail(e.message);
    throw e;
  }
}
```

```ts
import { integer, primaryKey, real, sqliteTable, text, unique } from 'drizzle-orm/sqlite-core';
import type { ChapterWarning, PublishStatus } from '@/lib/types';

const now = () => Date.now();

export const libraryRoot = sqliteTable('library_root', {
  id: integer('id').primaryKey({ autoIncrement: true }),
  path: text('path').notNull().unique(),
  createdAt: integer('created_at').notNull().$defaultFn(now),
});

export const novel = sqliteTable('novel', {
  id: integer('id').primaryKey({ autoIncrement: true }),
  name: text('name').notNull(),
  dirPath: text('dir_path').notNull().unique(),
  chapterSubdir: text('chapter_subdir').notNull().default(''),
  archived: integer('archived', { mode: 'boolean' }).notNull().default(false),
  dirMissing: integer('dir_missing', { mode: 'boolean' }).notNull().default(false),
  rootRemoved: integer('root_removed', { mode: 'boolean' }).notNull().default(false),
  createdAt: integer('created_at').notNull().$defaultFn(now),
});

export const chapter = sqliteTable(
  'chapter',
  {
    id: integer('id').primaryKey({ autoIncrement: true }),
    novelId: integer('novel_id').notNull().references(() => novel.id),
    seq: integer('seq').notNull(),
    title: text('title').notNull(),
    filePath: text('file_path').notNull(),
    wordCount: integer('word_count').notNull().default(0),
    contentHash: text('content_hash'),
    fileMissing: integer('file_missing', { mode: 'boolean' }).notNull().default(false),
    warning: text('warning').$type<ChapterWarning>(),
    mtimeMs: real('mtime_ms'),
    sizeBytes: integer('size_bytes'),
  },
  (t) => [unique().on(t.novelId, t.seq)],
);

export const scanIssue = sqliteTable('scan_issue', {
  id: integer('id').primaryKey({ autoIncrement: true }),
  novelId: integer('novel_id').notNull().references(() => novel.id),
  fileName: text('file_name').notNull(),
  kind: text('kind').$type<'unrecognized' | 'duplicate'>().notNull(),
  reason: text('reason').notNull(),
});

export const platform = sqliteTable('platform', {
  id: integer('id').primaryKey({ autoIncrement: true }),
  name: text('name').notNull().unique(),
  builtin: integer('builtin', { mode: 'boolean' }).notNull().default(false),
});

export const novelPlatform = sqliteTable(
  'novel_platform',
  {
    novelId: integer('novel_id').notNull().references(() => novel.id),
    platformId: integer('platform_id').notNull().references(() => platform.id),
    bookUrl: text('book_url'),
  },
  (t) => [primaryKey({ columns: [t.novelId, t.platformId] })],
);

export const publication = sqliteTable(
  'publication',
  {
    chapterId: integer('chapter_id').notNull().references(() => chapter.id),
    platformId: integer('platform_id').notNull().references(() => platform.id),
    status: text('status').$type<PublishStatus>().notNull().default('unpublished'),
    publishedAt: integer('published_at'),
    platformUrl: text('platform_url'),
    publishedHash: text('published_hash'),
  },
  (t) => [primaryKey({ columns: [t.chapterId, t.platformId] })],
);

export type Novel = typeof novel.$inferSelect;
export type Chapter = typeof chapter.$inferSelect;
export type Publication = typeof publication.$inferSelect;
export type Platform = typeof platform.$inferSelect;
```

```ts
// src/db/index.ts
import Database from 'better-sqlite3';
import { drizzle, type BetterSQLite3Database } from 'drizzle-orm/better-sqlite3';
import { migrate } from 'drizzle-orm/better-sqlite3/migrator';
import fs from 'node:fs';
import path from 'node:path';
import * as schema from './schema';
import { seedBuiltinPlatforms } from './seed';

export type Db = BetterSQLite3Database<typeof schema>;

/** file 传 ':memory:' 用于测试。会执行迁移与预置数据。 */
export function createDb(file: string): Db {
  if (file !== ':memory:') fs.mkdirSync(path.dirname(file), { recursive: true });
  const sqlite = new Database(file);
  sqlite.pragma('journal_mode = WAL');
  sqlite.pragma('foreign_keys = ON');
  const db = drizzle(sqlite, { schema });
  migrate(db, { migrationsFolder: path.join(process.cwd(), 'drizzle') });
  seedBuiltinPlatforms(db);
  return db;
}

const g = globalThis as unknown as { __novelHubDb?: Db };

/** Next 开发模式热更新下复用同一连接。 */
export function getDb(): Db {
  g.__novelHubDb ??= createDb(process.env.NOVEL_HUB_DB ?? path.join(process.cwd(), 'data', 'novel-hub.db'));
  return g.__novelHubDb;
}
```

```ts
import fs from 'node:fs';
import os from 'node:os';
import path from 'node:path';
import { createDb, type Db } from '@/db';
import { novel, type Novel } from '@/db/schema';

export function makeDb(): Db {
  return createDb(':memory:');
}

export function makeTmpDir(files: Record<string, string> = {}) {
  const dir = fs.realpathSync(fs.mkdtempSync(path.join(os.tmpdir(), 'novel-hub-')));
  for (const [rel, content] of Object.entries(files)) {
    const full = path.join(dir, rel);
    fs.mkdirSync(path.dirname(full), { recursive: true });
    fs.writeFileSync(full, content);
  }
  return { dir, cleanup: () => fs.rmSync(dir, { recursive: true, force: true }) };
}

export function insertNovel(db: Db, dirPath: string, over: Partial<typeof novel.$inferInsert> = {}): Novel {
  return db
    .insert(novel)
    .values({ name: path.basename(dirPath), dirPath, ...over })
    .returning()
    .get();
}
```

```tsx
  kind: 'success' | 'error' | 'warn';
  text: string;
  undo?: () => void | Promise<void>;
}
const Ctx = createContext<{ notify: (n: NoticeInput) => void }>({ notify: () => {} });
export const useNotice = () => useContext(Ctx);

const COLORS = {
  success: 'bg-emerald-50 border-emerald-300 text-emerald-900',
  error: 'bg-red-50 border-red-300 text-red-900',
  warn: 'bg-amber-50 border-amber-300 text-amber-900',
};

export function NoticeProvider({ children }: { children: React.ReactNode }) {
  const [n, setN] = useState<NoticeInput | null>(null);
  const notify = useCallback((x: NoticeInput) => setN(x), []);
  return (
    <Ctx.Provider value={{ notify }}>
      {n && (
        <div role="status" className={`mx-auto mt-4 flex max-w-6xl items-center gap-3 rounded border px-4 py-2 ${COLORS[n.kind]}`}>
          <span className="flex-1">{n.text}</span>
          {n.undo && (
            <button className="underline" onClick={async () => { await n.undo!(); setN(null); }}>
              撤销
            </button>
          )}
          <button aria-label="关闭" onClick={() => setN(null)}>×</button>
        </div>
      )}
      {children}
    </Ctx.Provider>
  );
}
```
